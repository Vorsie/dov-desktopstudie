# desktopstudie/qgis/layers.py
"""QGIS layers for one study: WMS/WCS layers from the catalogue and memory layers from the model."""
from __future__ import annotations

from pathlib import Path
from typing import Dict, Iterable, List, Sequence

from qgis.core import (Qgis, QgsCoordinateReferenceSystem, QgsDataSourceUri, QgsFeature, QgsField, QgsFields,
                       QgsGeometry, QgsLayerTreeGroup, QgsMarkerSymbol, QgsPalLayerSettings, QgsPointXY, QgsProject,
                       QgsRasterLayer, QgsTextFormat, QgsVectorFileWriter, QgsVectorLayer,
                       QgsVectorLayerSimpleLabeling, QgsFillSymbol, QgsLineSymbol)
from qgis.PyQt.QtCore import QVariant

from ..core.catalogue import MapEntry
from ..core.model import Borehole, Cpt, GwFilter, StudyZone

CRS = QgsCoordinateReferenceSystem("EPSG:31370")
POINT_STYLE = {  # kind -> (colour, marker)
    "sondering": ("#1f4e79", "circle"),
    "boring": ("#c00000", "square"),
    "peilput": ("#2e75b6", "triangle"),
}


def wms_layer(entry: MapEntry) -> QgsRasterLayer:
    uri = QgsDataSourceUri()
    uri.setParam("url", entry.wms_url)
    uri.setParam("layers", entry.wms_layer)
    uri.setParam("styles", "")
    uri.setParam("format", entry.image_format)
    uri.setParam("crs", "EPSG:31370")
    uri.setParam("dpiMode", "7")
    uri.setParam("contextualWMSLegend", "0")
    layer = QgsRasterLayer(str(uri.encodedUri(), "utf-8"), entry.title, "wms")
    if layer.isValid():
        layer.setOpacity(entry.opacity)
    return layer


def wcs_layer(url: str, coverage: str, name: str) -> QgsRasterLayer:
    uri = QgsDataSourceUri()
    uri.setParam("url", url)
    uri.setParam("identifier", coverage)
    uri.setParam("crs", "EPSG:31370")
    uri.setParam("format", "image/tiff")
    return QgsRasterLayer(str(uri.encodedUri(), "utf-8"), name, "wcs")


def _memory(geometry: str, name: str, fields: Sequence[tuple]) -> QgsVectorLayer:
    spec = "&".join([f"{geometry}?crs=EPSG:31370"] + [f"field={n}:{t}" for n, t in fields])
    layer = QgsVectorLayer(spec, name, "memory")
    return layer


def zone_layer(zone: StudyZone) -> QgsVectorLayer:
    layer = _memory("Polygon", "Onderzoekszone", [("naam", "string")])
    f = QgsFeature(layer.fields())
    f.setGeometry(QgsGeometry.fromPolygonXY([[QgsPointXY(x, y) for x, y in zone.ring]]))
    f["naam"] = zone.name
    layer.dataProvider().addFeatures([f])
    layer.updateExtents()
    layer.renderer().setSymbol(QgsFillSymbol.createSimple({"color": "255,0,0,30", "outline_color": "#ff0000", "outline_width": "0.8"}))
    return layer


def circle_layer(zone: StudyZone) -> QgsVectorLayer:
    from ..core import geometry as g

    layer = _memory("Polygon", f"Straal {zone.radius_m:.0f} m", [("straal_m", "double")])
    cx, cy = zone.centroid
    ring = g.buffer_point(cx, cy, zone.radius_m + max(g.bbox(zone.ring)[2] - cx, g.bbox(zone.ring)[3] - cy), n=96)
    f = QgsFeature(layer.fields())
    f.setGeometry(QgsGeometry.fromPolygonXY([[QgsPointXY(x, y) for x, y in ring]]))
    f["straal_m"] = zone.radius_m
    layer.dataProvider().addFeatures([f])
    layer.updateExtents()
    layer.renderer().setSymbol(QgsFillSymbol.createSimple({"color": "0,0,0,0", "outline_color": "#ff0000", "outline_style": "dash", "outline_width": "0.5"}))
    return layer


def line_layer(zone: StudyZone) -> QgsVectorLayer:
    layer = _memory("LineString", "Doorsnedelijn", [("naam", "string")])
    if zone.section_line:
        p, q = zone.section_line
        f = QgsFeature(layer.fields())
        f.setGeometry(QgsGeometry.fromPolylineXY([QgsPointXY(*p), QgsPointXY(*q)]))
        f["naam"] = "A-A'"
        layer.dataProvider().addFeatures([f])
        layer.updateExtents()
    layer.renderer().setSymbol(QgsLineSymbol.createSimple({"color": "#000000", "width": "0.6"}))
    return layer


POINT_FIELDS = [("nummer", "string"), ("afstand_m", "double"), ("diepte_m", "double"), ("datum", "string"),
                ("methode", "string"), ("z_mtaw", "double"), ("url", "string")]


def points_layer(kind: str, items: Iterable) -> QgsVectorLayer:
    names = {"sondering": "Sonderingen", "boring": "Boringen", "peilput": "Peilputten"}
    layer = _memory("Point", names[kind], POINT_FIELDS)
    feats = []
    for it in items:
        f = QgsFeature(layer.fields())
        f.setGeometry(QgsGeometry.fromPointXY(QgsPointXY(it.x, it.y)))
        number = getattr(it, "number", None) or f"{getattr(it, 'gw_id', '?')}/{getattr(it, 'filter_no', '')}"
        depth = getattr(it, "depth_m", None) if hasattr(it, "depth_m") else getattr(it, "filter_base_m", None)
        f.setAttributes([number, float(it.distance_m), depth, getattr(it, "date", None), getattr(it, "method", None),
                         getattr(it, "z_mtaw", None), it.url])
        feats.append(f)
    layer.dataProvider().addFeatures(feats)
    layer.updateExtents()
    colour, marker = POINT_STYLE[kind]
    layer.renderer().setSymbol(QgsMarkerSymbol.createSimple({"name": marker, "color": colour, "size": "2.6",
                                                             "outline_color": "white", "outline_width": "0.3"}))
    settings = QgsPalLayerSettings()
    settings.fieldName = "nummer"
    fmt = QgsTextFormat()
    fmt.setSize(7)
    settings.setFormat(fmt)
    layer.setLabeling(QgsVectorLayerSimpleLabeling(settings))
    layer.setLabelsEnabled(True)
    return layer


def add_group(project: QgsProject, title: str, layers: Sequence, visible: bool = True) -> QgsLayerTreeGroup:
    root = project.layerTreeRoot()
    group = root.addGroup(title)
    for layer in layers:
        project.addMapLayer(layer, False)
        node = group.addLayer(layer)
        node.setItemVisibilityChecked(visible)
    return group


def write_geopackage(layers: Sequence[QgsVectorLayer], path: Path) -> None:
    path = Path(path)
    for i, layer in enumerate(layers):
        opts = QgsVectorFileWriter.SaveVectorOptions()
        opts.driverName = "GPKG"
        opts.layerName = layer.name()
        opts.actionOnExistingFile = (QgsVectorFileWriter.CreateOrOverwriteFile if i == 0
                                     else QgsVectorFileWriter.CreateOrOverwriteLayer)
        err, msg, *_ = QgsVectorFileWriter.writeAsVectorFormatV3(layer, str(path), project_transform_context(), opts)
        if err != QgsVectorFileWriter.NoError:
            raise RuntimeError(f"GeoPackage schrijven mislukt voor {layer.name()}: {msg}")


def project_transform_context():
    return QgsProject.instance().transformContext()
```

# desktopstudie/qgis/dem.py
"""Relief of the zone from the DHMV II DTM (WCS): min / max / mean in mTAW."""
from __future__ import annotations

from typing import Optional, Tuple

from qgis.analysis import QgsZonalStatistics
from qgis.core import QgsVectorLayer

from ..core.catalogue import DHMV_WCS_COVERAGE, DHMV_WCS_URL
from . import layers


def relief_of_zone(zone_layer: QgsVectorLayer, log=None) -> Optional[Tuple[float, float, float]]:
    dtm = layers.wcs_layer(DHMV_WCS_URL, DHMV_WCS_COVERAGE, "DHMV II DTM 1 m")
    if not dtm.isValid():
        if log:
            log.warning("DHMV WCS niet beschikbaar; geen reliefstatistiek")
        return None
    stats = QgsZonalStatistics.Statistic.Min | QgsZonalStatistics.Statistic.Max | QgsZonalStatistics.Statistic.Mean
    zs = QgsZonalStatistics(zone_layer, dtm, "dhmv_", 1, stats)
    zs.calculateStatistics(None)
    feat = next(zone_layer.getFeatures())
    try:
        lo, hi, mean = float(feat["dhmv_min"]), float(feat["dhmv_max"]), float(feat["dhmv_mean"])
    except (KeyError, TypeError, ValueError):
        if log:
            log.warning("DHMV zonale statistiek leverde geen waarden")
        return None
    return lo, hi, mean
```

"""Builds one multi-page QgsPrintLayout from the core's Report tree."""
from __future__ import annotations

import datetime as dt
from pathlib import Path
from typing import Dict, List, Sequence

from qgis.core import (Qgis, QgsCoordinateReferenceSystem, QgsLayoutFrame, QgsLayoutItemLabel, QgsLayoutItemLegend,
                       QgsLayoutItemMap, QgsLayoutItemPage, QgsLayoutItemPicture, QgsLayoutItemScaleBar,
                       QgsLayoutItemTextTable, QgsLayoutMultiFrame, QgsLayoutTable, QgsLayoutTableColumn,
                       QgsPrintLayout, QgsProject, QgsRectangle)
from qgis.PyQt.QtGui import QFont

from ..core import catalogue, geometry
from ..core.report_content import Chapter, FigurePage, MapPage, Report, TablePage, TextPage
from .compat import point_mm, size_mm

MARGIN = 15.0
MAP_W, MAP_H = 180.0, 200.0
NORTH_ARROW = Path(__file__).resolve().parents[1] / "resources" / "noordpijl.svg"


class LayoutBuilder:
    def __init__(self, project: QgsProject, report: Report, layers_by_map: Dict[str, list], overlays: Dict[str, list],
                 out_dir: Path, zone_ring: Sequence, meta: dict):
        """layers_by_map: map_id -> [QgsMapLayer,...] to draw (WMS + basemap); overlays: keys 'zone', 'investigations',
        'section' -> memory layers drawn on top when the page asks for them."""
        self.project, self.report, self.layers_by_map, self.overlays = project, report, layers_by_map, overlays
        self.out_dir, self.zone_ring, self.meta = Path(out_dir), list(zone_ring), meta
        self.layout = QgsPrintLayout(project)
        self.layout.initializeDefaults()
        self.layout.setName("DOV Desktopstudie")
        self.page_index = -1

    # --- pages -------------------------------------------------------------------------------
    def new_page(self) -> int:
        if self.page_index >= 0:
            page = QgsLayoutItemPage(self.layout)
            page.setPageSize("A4", QgsLayoutItemPage.Orientation.Portrait)
            self.layout.pageCollection().addPage(page)
        self.page_index += 1
        return self.page_index

    def label(self, text: str, x: float, y: float, w: float, h: float, size: float = 9, html: bool = False,
              frame: bool = False, bold: bool = False) -> QgsLayoutItemLabel:
        item = QgsLayoutItemLabel(self.layout)
        if html:
            item.setMode(QgsLayoutItemLabel.Mode.ModeHtml)
        item.setText(text)
        font = QFont("Arial", int(size))
        font.setBold(bold)
        item.setFont(font)
        item.setFrameEnabled(frame)
        self.layout.addLayoutItem(item)
        item.attemptMove(point_mm(x, y), page=self.page_index)
        item.attemptResize(size_mm(w, h))
        return item

    def header(self, chapter: Chapter, title: str) -> None:
        self.label(f"{chapter.number}. {chapter.title}", MARGIN, 10, 180, 8, size=13, bold=True)
        self.label(title, MARGIN, 18, 180, 7, size=10)

    def footer(self) -> None:
        text = f"{self.meta.get('company', '')} - {self.meta.get('project', '')} - pagina [% @layout_page %] / [% @layout_numpages %]"
        self.label(text, MARGIN, 285, 180, 6, size=7)

    # --- map page ----------------------------------------------------------------------------
    def map_extent(self, scale: int, extent_factor: float) -> QgsRectangle:
        minx, miny, maxx, maxy = geometry.bbox(self.zone_ring)
        cx, cy = (minx + maxx) / 2, (miny + maxy) / 2
        ratio = MAP_H / MAP_W
        w_scale = MAP_W / 1000.0 * scale            # metres shown at the target scale
        w_needed = max(maxx - minx, (maxy - miny) / ratio) * max(extent_factor, 1.0)
        w = max(w_scale, w_needed)
        h = w * ratio
        return QgsRectangle(cx - w / 2, cy - h / 2, cx + w / 2, cy + h / 2)

    def map_page(self, chapter: Chapter, page: MapPage) -> None:
        self.new_page()
        self.header(chapter, page.title)
        entry = catalogue.by_id(page.map_id)
        layers = list(self.layers_by_map.get(page.map_id, []))
        if page.show_investigations:
            layers = self.overlays.get("investigations", []) + layers
        if page.show_section_line:
            layers = self.overlays.get("section", []) + layers
        layers = self.overlays.get("zone", []) + layers
        m = QgsLayoutItemMap(self.layout)
        m.setCrs(QgsCoordinateReferenceSystem("EPSG:31370"))
        m.setLayers(layers)
        m.setFrameEnabled(True)
        self.layout.addLayoutItem(m)
        m.attemptMove(point_mm(MARGIN, 30), page=self.page_index)
        m.attemptResize(size_mm(MAP_W, MAP_H))
        m.setExtent(self.map_extent(page.scale, page.extent_factor))
        real_scale = int(round(m.scale()))
        # info boxes
        self.label(f"<b>{self.meta.get('project', '')}</b><br>{self.meta.get('created_at', '')[:10]}<br>{entry.title}<br>schaal 1:{real_scale:,}".replace(",", " "),
                   140, 46, 55, 22, size=7, html=True, frame=True)
        self.label(f"{entry.attribution}<br>{entry.licence}<br>opgehaald {dt.date.today().isoformat()}", 140, 210, 55, 18, size=6, html=True, frame=True)
        # north arrow
        arrow = QgsLayoutItemPicture(self.layout)
        arrow.setPicturePath(str(NORTH_ARROW))
        self.layout.addLayoutItem(arrow)
        arrow.attemptMove(point_mm(183, 32), page=self.page_index)
        arrow.attemptResize(size_mm(12, 12))
        # scale bar
        bar = QgsLayoutItemScaleBar(self.layout)
        bar.setLinkedMap(m)
        bar.setStyle("Single Box")
        bar.setUnits(Qgis.DistanceUnit.Meters)
        bar.setUnitLabel("m")
        bar.setNumberOfSegments(2)
        bar.setNumberOfSegmentsLeft(0)
        bar.setUnitsPerSegment(max(50.0, round(real_scale / 20.0 / 50.0) * 50.0))
        bar.applyDefaultSize()
        self.layout.addLayoutItem(bar)
        bar.attemptMove(point_mm(MARGIN, 232), page=self.page_index)
        # legend
        if page.legend and layers:
            legend = QgsLayoutItemLegend(self.layout)
            legend.setLinkedMap(m)
            legend.setAutoUpdateModel(False)
            legend.setTitle("Legenda")
            root = legend.model().rootGroup()
            root.clear()
            for lyr in self.layers_by_map.get(page.map_id, []):
                root.addLayer(lyr)
            self.layout.addLayoutItem(legend)
            legend.attemptMove(point_mm(MARGIN, 240), page=self.page_index)
            legend.attemptResize(size_mm(120, 42))
        if page.note:
            self.label(page.note, MARGIN, 24, 180, 5, size=7)
        self.footer()

    # --- other pages -------------------------------------------------------------------------
    def figure_page(self, chapter: Chapter, page: FigurePage) -> None:
        self.new_page()
        self.header(chapter, page.title)
        pic = QgsLayoutItemPicture(self.layout)
        pic.setPicturePath(str(self.out_dir / page.image_path))
        pic.setResizeMode(QgsLayoutItemPicture.ResizeMode.Zoom)
        self.layout.addLayoutItem(pic)
        pic.attemptMove(point_mm(MARGIN, 30), page=self.page_index)
        pic.attemptResize(size_mm(180, 235))
        if page.caption:
            self.label(page.caption, MARGIN, 268, 180, 12, size=7)
        self.footer()

    def table_page(self, chapter: Chapter, page: TablePage) -> None:
        self.new_page()
        self.header(chapter, page.title)
        table = QgsLayoutItemTextTable(self.layout)
        self.layout.addMultiFrame(table)
        table.setColumns([QgsLayoutTableColumn(h) for h in page.columns])
        table.setContents([[str(c) for c in row] for row in page.rows])
        table.setHeaderMode(QgsLayoutTable.HeaderMode.AllFrames)
        table.setResizeMode(QgsLayoutMultiFrame.ResizeMode.ExtendToNextPage)
        table.setContentFont(QFont("Arial", 7))
        table.setHeaderFont(QFont("Arial", 7, QFont.Weight.Bold))
        frame = QgsLayoutFrame(self.layout, table)
        frame.attemptMove(point_mm(MARGIN, 30), page=self.page_index)
        frame.attemptResize(size_mm(180, 245))
        table.addFrame(frame)
        if page.note:
            self.label(page.note, MARGIN, 24, 180, 5, size=7)
        self.footer()

    def text_page(self, chapter: Chapter, page: TextPage) -> None:
        self.new_page()
        self.header(chapter, page.title)
        self.label(page.html, MARGIN, 30, 180, 245, size=9, html=True)
        self.footer()

    def title_page(self) -> None:
        self.new_page()
        logo = self.meta.get("logo_path")
        y = 20.0
        if logo and Path(logo).exists():
            pic = QgsLayoutItemPicture(self.layout)
            pic.setPicturePath(logo)
            pic.setResizeMode(QgsLayoutItemPicture.ResizeMode.Zoom)
            self.layout.addLayoutItem(pic)
            pic.attemptMove(point_mm(MARGIN, y), page=self.page_index)
            pic.attemptResize(size_mm(60, 30))
            y += 35
        self.label(self.report.title, MARGIN, y, 180, 14, size=20, bold=True)
        rows = [("Project", self.meta.get("project")), ("Projectnummer", self.meta.get("project_number")),
                ("Adres", self.meta.get("address")), ("Gemeente", self.meta.get("municipality")),
                ("Zone", self.meta.get("zone_name")), ("Datum", self.meta.get("created_at", "")[:10]),
                ("Auteur", self.meta.get("author")), ("Bedrijf", self.meta.get("company"))]
        html = "<table>" + "".join(f"<tr><td><b>{k}</b></td><td>{v or '-'}</td></tr>" for k, v in rows) + "</table>"
        self.label(html, MARGIN, y + 20, 180, 60, size=9, html=True)
        toc = "<b>Inhoud</b><br>" + "<br>".join(f"{c.number}. {c.title}" for c in self.report.chapters)
        self.label(toc, MARGIN, y + 85, 180, 60, size=9, html=True)
        self.label(self.meta.get("disclaimer", ""), MARGIN, 240, 180, 30, size=7, html=True)

    def build(self) -> QgsPrintLayout:
        self.title_page()
        for chapter in self.report.chapters:
            for page in chapter.pages:
                if isinstance(page, MapPage):
                    self.map_page(chapter, page)
                elif isinstance(page, FigurePage):
                    self.figure_page(chapter, page)
                elif isinstance(page, TablePage):
                    self.table_page(chapter, page)
                else:
                    self.text_page(chapter, page)
        return self.layout


def build_layout(project, report, layers_by_map, overlays, out_dir, zone_ring, meta) -> QgsPrintLayout:
    return LayoutBuilder(project, report, layers_by_map, overlays, out_dir, zone_ring, meta).build()

from __future__ import annotations

from pathlib import Path
from typing import List

from qgis.core import QgsLayoutExporter, QgsPrintLayout, QgsProject


def export_pdf(layout: QgsPrintLayout, path: Path, dpi: int = 150) -> Path:
    settings = QgsLayoutExporter.PdfExportSettings()
    settings.dpi = dpi
    settings.rasterizeWholeImage = False
    result = QgsLayoutExporter(layout).exportToPdf(str(path), settings)
    if result != QgsLayoutExporter.ExportResult.Success:
        raise RuntimeError(f"PDF-export mislukt: {result}")
    return Path(path)


def export_pages_png(layout: QgsPrintLayout, directory: Path, dpi: int = 72) -> List[Path]:
    directory = Path(directory)
    directory.mkdir(parents=True, exist_ok=True)
    settings = QgsLayoutExporter.ImageExportSettings()
    settings.dpi = dpi
    result = QgsLayoutExporter(layout).exportToImage(str(directory / "pagina.png"), settings)
    if result != QgsLayoutExporter.ExportResult.Success:
        raise RuntimeError(f"PNG-export mislukt: {result}")
    return sorted(directory.glob("pagina_*.png"))


def write_project(project: QgsProject, path: Path) -> Path:
    if not project.write(str(path)):
        raise RuntimeError(f"project schrijven mislukt: {path}")
    return Path(path)

```python
# desktopstudie/qgis/pipeline.py
"""One study end-to-end: core run -> relief -> checks -> report -> layers -> layout -> exports."""
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Callable, Dict, List, Optional

from qgis.core import QgsProject

from ..core import catalogue, checks
from ..core.logging_util import Log
from ..core.model import StudyResult, StudyZone
from ..core.report_content import Report, ReportMeta, build_report
from ..core.services.http import HttpClient
from ..core.study import Settings, run
from . import dem, export, layers, layout as layout_mod


@dataclass
class PipelineResult:
    result: StudyResult
    report: Report
    pdf: Path
    project_file: Path
    geopackage: Path
    page_pngs: List[Path]


CHAPTER_GROUPS = {"ligging": "1 Ligging en topografie", "historisch": "2 Historische kaarten", "geologie": "3 Geologie en bodem"}


def run_core(zone: StudyZone, settings: Settings, out_dir: Path, log: Log, progress=None, should_cancel=None,
             cache_mode: str = "use") -> StudyResult:
    client = HttpClient(cache_dir=out_dir / "data" / "cache", log=log.child("http"), cache_mode=cache_mode)
    return run(zone, settings, client, out_dir, progress=progress, log=log.child("study"), should_cancel=should_cancel)


def finish(project: QgsProject, result: StudyResult, meta: ReportMeta, out_dir: Path, log: Log,
           progress: Optional[Callable[[float, str], None]] = None) -> PipelineResult:
    """Main-thread part: relief, checks, report, layers, layout, exports."""
    out_dir = Path(out_dir)
    p = progress or (lambda f, m: None)
    zone_layer = layers.zone_layer(result.zone)
    p(0.05, "Relief uit DHMV")
    result.relief = dem.relief_of_zone(zone_layer, log=log.child("dem"))
    result.signaleringen = checks.run_all(result) + [s for s in result.signaleringen if s.code in ("wfs_afgekapt", "doorsnede_onvolledig")]
    result.write_json(out_dir / "data" / "studie.json")
    report = build_report(result, meta)
    p(0.15, "Lagen")
    overlays = {"zone": [zone_layer], "section": [layers.line_layer(result.zone)],
                "investigations": [layers.points_layer("sondering", result.cpts), layers.points_layer("boring", result.boreholes),
                                   layers.points_layer("peilput", result.gw_filters), layers.circle_layer(result.zone)]}
    layers_by_map: Dict[str, list] = {}
    for chapter, group_title in CHAPTER_GROUPS.items():
        group_layers = []
        for entry in catalogue.entries(chapter):
            lyr = layers.wms_layer(entry)
            if not lyr.isValid():
                log.warning(f"WMS-laag niet geldig: {entry.id}")
                continue
            layers_by_map[entry.id] = [lyr]
            group_layers.append(lyr)
        layers.add_group(project, group_title, group_layers, visible=False)
    layers.add_group(project, "4 Onderzoekszone en doorsnede", overlays["zone"] + overlays["section"])
    layers.add_group(project, "5 Grondonderzoek DOV", overlays["investigations"])
    p(0.35, "Layout")
    lay = layout_mod.build_layout(project, report, layers_by_map, overlays, out_dir, result.zone.ring,
                                  report.meta | {"logo_path": meta.logo_path})
    project.layoutManager().addLayout(lay)
    p(0.55, "PDF")
    pdf = export.export_pdf(lay, out_dir / "rapport.pdf")
    p(0.85, "Pagina's als PNG")
    pngs = export.export_pages_png(lay, out_dir / "paginas")
    gpkg = out_dir / "data" / "studie.gpkg"
    layers.write_geopackage(overlays["zone"] + overlays["section"] + overlays["investigations"], gpkg)
    project_file = export.write_project(project, out_dir / "studie.qgz")
    p(1.0, "Klaar")
    return PipelineResult(result, report, pdf, project_file, gpkg, pngs)


def run_pipeline(zone, settings, meta, out_dir, project, log, progress=None, should_cancel=None, cache_mode="use") -> PipelineResult:
    result = run_core(zone, settings, Path(out_dir), log, progress, should_cancel, cache_mode)
    return finish(project, result, meta, Path(out_dir), log, progress)
```

"""Draw a polygon or a line on the canvas: left click adds a vertex, right click finishes."""
from __future__ import annotations

from typing import Callable, List, Tuple

from qgis.core import Qgis, QgsPointXY, QgsWkbTypes
from qgis.gui import QgsMapTool, QgsRubberBand
from qgis.PyQt.QtCore import Qt
from qgis.PyQt.QtGui import QColor


class DrawTool(QgsMapTool):
    def __init__(self, canvas, geometry_type, on_done: Callable[[List[Tuple[float, float]]], None]):
        super().__init__(canvas)
        self.canvas, self.on_done = canvas, on_done
        self.band = QgsRubberBand(canvas, geometry_type)
        self.band.setColor(QColor(255, 0, 0, 120))
        self.band.setWidth(2)
        self.points: List[QgsPointXY] = []

    def canvasReleaseEvent(self, event):
        if event.button() == Qt.MouseButton.RightButton:
            pts = [(p.x(), p.y()) for p in self.points]
            self.band.reset(self.band.geometryType()) if hasattr(self.band, "geometryType") else self.band.reset()
            self.points = []
            self.on_done(pts)
            return
        p = self.toMapCoordinates(event.pos())
        self.points.append(p)
        self.band.addPoint(p, True)


def polygon_tool(canvas, on_done) -> DrawTool:
    return DrawTool(canvas, Qgis.GeometryType.Polygon, on_done)


def line_tool(canvas, on_done) -> DrawTool:
    return DrawTool(canvas, Qgis.GeometryType.Line, on_done)
```

"""Background task: the core runs in a worker thread; the QGIS part runs on the main thread afterwards."""
from __future__ import annotations

        self.zone, self.settings, self.meta, self.out_dir, self.cache_mode = zone, settings, meta, Path(out_dir), cache_mode
        self.on_finished, self.result, self.error = on_finished, None, None
        self.log = Log("plugin", sink=lambda m: QgsMessageLog.logMessage(m, "DOV Desktopstudie", Qgis.MessageLevel.Info), scope="qgis")

    def run(self) -> bool:
        try:
            self.result = pipeline.run_core(self.zone, self.settings, self.out_dir, self.log,
                                            progress=lambda f, m: self.setProgress(f * 100), should_cancel=self.isCanceled,
                                            cache_mode=self.cache_mode)
            return True
        except Exception as exc:  # noqa: BLE001 - reported in finished()
            self.error = exc
            return False

    def finished(self, ok: bool) -> None:
        if not ok or self.result is None:
            self.on_finished(None, self.error)
            return
        try:
            out = pipeline.finish(QgsProject.instance(), self.result, self.meta, self.out_dir, self.log)
            self.on_finished(out, None)
        except Exception as exc:  # noqa: BLE001
            self.on_finished(None, exc)
```
